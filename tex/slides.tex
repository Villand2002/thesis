\begin{frame}{問題設定}
一つの大学における割当を考える.
\begin{itemize}
  \item 一般カテゴリ $o$ と複数の留保カテゴリ $T$
  \item 学生 $i$ は点数 $\sigma(i)$ と資格集合 $V(i)$ を持つ(全員が $o$ の資格を持つ)
  \item \alert{点数制約:一般カテゴリの cutoff $e_o(Y)$ と留保カテゴリ合格者の点数の差を $\delta$ 以内に抑える}
  \item 研究課題:reserved category protection・公平性・効率性・耐戦略性はどこまで両立するか
\end{itemize}
\end{frame}

% 先行研究
% =============================================================
\begin{frame}{先行研究 1:Hafalir, Yenmez, Yildirim (2013)}
\small
\textbf{問題意識}\quad
多数派学生の人数に上限を課す方式(majority quota)は,
かえって少数派学生の結果を悪化させることがある.

\medskip
\textbf{提案:minority reserve}
\begin{itemize}
  \item 少数派のために定員を留保するが,\alert{少数派の応募が足りなければ,その定員は多数派にも開放される}
  \item 留保は「絶対に確保しておく定員」ではなく,需要がなければ開放される
        $\Rightarrow$ この性質を soft reserve と呼ぶ
  \item 一方,majority quota は超えられない上限であり,\textbf{hard} である
\end{itemize}

\medskip
\textbf{結果}\quad minority reserve の下での DA は,耐戦略性を満たし安定である.

\medskip
\begin{block}{本研究との相違点}
  留保カテゴリの未使用の定員を開放するという発想は,本研究の柔軟カテゴリ(後述)と同じである.
  ただし,\textbf{留保カテゴリの合格者の点数を一般カテゴリと比べて制約する条件はない}.
\end{block}
\end{frame}

\begin{frame}{先行研究 2:Aziz and Sun (2021; 2025)}
\small
\textbf{設定}
\begin{itemize}
  \item 学生は\textbf{複数のタイプ}(カテゴリ)に同時に属しうる
  \item 合格した学生は,どれか\textbf{一つ}のタイプの定員だけを消費する(one-to-one counting)
  \item 学校は多段階(multi-rank)の多様性目標を持つ(例:最低限確保したい人数と,できれば確保したい人数)
\end{itemize}

\medskip
\textbf{提案}\quad 学校の選抜ルール(choice function).次の三つの性質を満たすルールはこれだけであることを示した.
\begin{itemize}
  \item \textbf{maximal diversity}:\alert{多様性目標をできるだけ多く満たす}学生集合を選ぶ
  \item 非浪費性,no justified envy
\end{itemize}
\end{frame}

\begin{frame}{先行研究 3:Aygün and Turhan (2023)}
\small
\textbf{背景}\quad インド最高裁(2008)は,OBC(その他後進諸階級)カテゴリの定員が埋まらなければ一般カテゴリに戻せと命じたが,
\alert{具体的な実施方法は定めなかった}.その結果,手続が統一されないまま運用されてきた.

\medskip
\textbf{提案:Backward Transfers(BT)選抜ルールと DA-BT}
\begin{enumerate}
  \item まず一般カテゴリを点数順に埋め,次に各留保カテゴリを埋める
  \item OBC に未使用の定員が残れば,その人数を一般カテゴリの定員に移して選抜をやり直す
  \item 未使用の定員がなくなるまで繰り返す
\end{enumerate}
法的要件と政策目標を公理として定式化し,それらを同時に満たすメカニズムは DA-BT だけであることを示した.

\medskip
一般カテゴリを先に埋めるため,\textbf{一般カテゴリの合格者は留保カテゴリの合格者より高得点}になる.
つまり「一般 cutoff $\geq$ 留保 cutoff」が保証される.
しかし\alert{cutoff 差の大きさ($\delta$)は扱っていない}.
\end{frame}

\begin{frame}{基本記号}
\small
\begin{itemize}
  \item 学生集合 $I$,留保カテゴリ集合 $T$,一般カテゴリ $o$
  \item $V := T \cup \{o\}$,定員 $q_c \in \mathbb{N}\ (c \in V)$,総定員 $Q := \sum_{c \in V} q_c$
  \item 許容される cutoff の差 $\delta \geq 0$
\end{itemize}
各学生 $i \in I$ について
\begin{itemize}
  \item 点数 $\sigma(i) \geq 0$,$\ i \neq j \Rightarrow \sigma(i) \neq \sigma(j)$
  \item 資格カテゴリ集合 $V(i) \subseteq V$,$\ o \in V(i)$
  \item $X_i$ 上の厳密な選好 $\succ_i$($\emptyset$ は外部オプション)
  \item 第一希望:$\topf_i$($V(i) \cup \{\emptyset\}$ の中で $\succ_i$ について最も好むもの)
\end{itemize}
\end{frame}

\begin{frame}{契約とマッチング}
\small
\begin{block}{Definition(契約)}
  \[
    X := \{\, (i,c) : i \in I,\ c \in V(i) \cup \{\emptyset\} \,\}
  \]
  $(i,c)$:学生 $i$ がカテゴリ $c$ で合格.$(i,\emptyset)$:割当なし.\\
  $Y \subseteq X$ に対し
  \[
    Y_i := \{\, (i,c) \in Y \,\}, \qquad Y_c := \{\, (i,c) \in Y \,\}
  \]
\end{block}
\begin{block}{Definition(マッチング)}
  $Y \subseteq X$ がマッチング $\iff$ $\forall i \in I:\ |Y_i| = 1$ かつ $Y_i \succeq_i \emptyset$
\end{block}
$|Y_c|$ はカテゴリ $c$ に割り当てられた学生の人数.
\end{frame}

\begin{frame}{一般カテゴリの cutoff}
\begin{block}{Definition(不合格者と cutoff)}
  \[
    R_o(Y) := \{\, i \in I : (i,\emptyset) \in Y,\ o \succ_i \emptyset \,\}
  \]
  ($o$ を外部オプションより好むのに,外部オプションに割り当てられた学生)
  \[
    e_o(Y) :=
    \begin{cases}
      \max_{i \in R_o(Y)} \sigma(i) & |R_o(Y)| \neq 0 \\
      0 & |R_o(Y)| = 0
    \end{cases}
  \]
\end{block}
$e_o(Y)$:$o$ を希望しながら不合格になった学生の最高点.
一般カテゴリの cutoff として使う.
\end{frame}

\begin{frame}{固定カテゴリの quota feasibility と score-feasibility}
\begin{block}{Definition(Quota feasibility,固定カテゴリ)}
  \[
    Y \text{ は quota feasible} \iff |Y_c| \leq q_c \quad (\forall c \in V)
  \]
\end{block}
\begin{block}{Definition(Score-feasibility)}
  $Y$ が score-feasible であるとは,次が成り立つこと:
  \[
    \forall (i,c) \in Y \text{ s.t. } c \in T:\ \ \sigma(i) \geq e_o(Y) - \delta
  \]
\end{block}
\begin{itemize}
  \item $o$ の定員も $q_o$ に固定され,留保カテゴリの未使用の定員は誰にも回らない
  \item 誰も落ちていない間は $e_o = 0$ なので,制約は常に満たされる.SSD の実行中は
        最初の脱落者の点数で $e_o$ が確定し,以後変化しない
\end{itemize}
\end{frame}

\begin{frame}{公平性と耐戦略性}
\small
\begin{block}{Definition(公平性)}
  マッチング $Y$ において,学生 $i$ が $j$ に \textbf{justified envy} を持つ $\iff$
  \[
    Y_j = \{(j, c)\}, \qquad c \in V(i), \qquad c \succ_i Y_i, \qquad \sigma(i) > \sigma(j)
  \]
  どの学生も justified envy を持たないとき,$Y$ は公平である.
\end{block}
\begin{block}{Definition(耐戦略性)}
  メカニズム $\varphi$ は選好プロファイル $\succ = (\succ_i)_{i\in I}$ にマッチングを対応させる.
  $\varphi$ が耐戦略的 $\iff$
  \[
    \forall i \in I,\ \forall \succ,\ \forall \succ'_i:\quad
    \varphi_i(\succ) \ \succeq_i\ \varphi_i(\succ'_i, \succ_{-i})
  \]
  (どの学生も,選好を偽って厳密に得をすることはできない)
\end{block}
\end{frame}

\begin{frame}{効率性:非浪費性と Pareto 効率性}
\small
\begin{block}{Definition(非浪費性)}
  マッチング $Y$ において,学生 $i$ がカテゴリ $c$ を \textbf{claim する} $\iff$
  \[
    c \in V(i), \qquad c \succ_i Y_i,
  \]
  かつ $(Y \setminus Y_i) \cup \{(i,c)\}$ が quota feasible かつ score-feasible.\\
  claim する学生が存在しないとき,$Y$ は非浪費的である.
\end{block}
\begin{block}{Definition(Pareto 効率性, PE)}
  $Y'$ が $Y$ を \textbf{Pareto 支配する} $\iff$
  \[
    \forall i \in I:\ Y'_i \succeq_i Y_i
    \quad\text{かつ}\quad
    \exists j \in I:\ Y'_j \succ_j Y_j
  \]
  quota feasible かつ score-feasible なマッチング $Y$ が PE $\iff$
  quota feasible かつ score-feasible な他のマッチングに Pareto 支配されない.
\end{block}
PE ならば非浪費的である.
\end{frame}

\begin{frame}{SSD と定理 1:固定カテゴリでは全部成り立つ}
\begin{block}{Definition(Score-based Serial Dictatorship, SSD)}
  点数の高い順に学生を選び,割当後の(部分)マッチングが
  quota feasible かつ score-feasible となるカテゴリ(外部オプション含む)のうち,
  最も好むものへ割り当てる.
\end{block}
\begin{block}{Theorem 1}
  固定カテゴリモデルにおいて,SSD は quota feasibility・score-feasibility・公平性・
  PE・\textbf{耐戦略性}をすべて満たす.
\end{block}
\medskip
\alert{柔軟カテゴリにすると,これらを同時には満たせなくなる}
\end{frame}

% =============================================================
\begin{frame}{柔軟カテゴリの quota feasibility}
\begin{block}{Definition(Quota feasibility,柔軟カテゴリ)}
  $Y$ が quota feasible であるとは,次が成り立つこと:
  \[
    |Y_c| \leq q_c \ \ (\forall c \in T)
    \quad\text{かつ}\quad
    \sum_{c \in V} |Y_c| \leq Q
  \]
\end{block}
\begin{itemize}
  \item $o$ 単独の上限はない
  \item 留保カテゴリの未使用の定員の分だけ $o$ の定員を増やせる(de-reservation)
  \item $e_o(Y)$ と score-feasibility の定義は固定カテゴリと同じ
\end{itemize}
\medskip
% \alert{問題:未使用の定員を無制限に $o$ に回すと,留保カテゴリを望む学生が締め出されうる}\\
% $\Rightarrow$ 留保カテゴリの定員をどこまで守るかを定める条件が reserved category protection
\end{frame}

\begin{frame}{Reserved category protection:定義}
\small
\begin{block}{Definition(保護水準)}
  \[
    \dtop{c} := \min\bigl\{\, q_c,\ |\{\, i \in I : \topf_i = c \,\}| \,\bigr\}
    \qquad (c \in T)
  \]
  $c$ の定員のうち,$o$ に回さず $c$ のために残す人数
  ($c$ を第一希望とする学生の人数分)
\end{block}
\begin{block}{Definition(Reserved category protection)}
  $Y$ が reserved category protection を満たすとは,次が成り立つこと:
  \[
    |Y_o| \;\leq\; q_o + \sum_{c \in T} \bigl( q_c - \max\{\, |Y_c|,\ \dtop{c} \,\} \bigr)
  \]
\end{block}
$r_c(Y) := q_c - \max\{\, |Y_c|,\ \dtop{c} \,\}$($c$ から $o$ に回せる人数)とおくと,
\begin{itemize}
  \item $|Y_c| \geq \dtop{c}$ のとき:$r_c = q_c - |Y_c|$
        $\Rightarrow$ \textbf{未使用の定員はすべて} $o$ に回してよい
  \item $|Y_c| < \dtop{c}$ のとき:$r_c = q_c - \dtop{c}$
        $\Rightarrow$ $\dtop{c} - |Y_c|$ 人分は\textbf{未使用のまま残す}
\end{itemize}
\end{frame}

\begin{frame}{柔軟カテゴリでの cutoff の定義}
cutoff の定義には二通りが考えられる.
\begin{itemize}
  \item \textbf{合格者側}:$\min\{\, \sigma(i) : (i,o) \in Y \,\}$($o$ の合格者の最低点)
  \item \textbf{不合格者側(本研究)}:$e_o(Y) = \max_{i \in R_o(Y)} \sigma(i)$
\end{itemize}
\medskip
柔軟カテゴリでは $o$ の定員が固定されていない.
留保カテゴリの定員を $o$ に回すと $o$ の合格者が増え,合格者の最低点は\textbf{下がる}.
\begin{itemize}
  \item 合格者側の cutoff は,最後に $o$ に何人入るかが決まるまで確定しない
  \item SSD で学生 $i$ を $t$ に入れてよいかをその時点で判定しても,
        後で $o$ の合格者が増えると判定が変わりうる
\end{itemize}
\end{frame}

\begin{frame}{制約付き非浪費性と制約付き PE}
\small
次の Theorem 2 で見るように,reserved category protection を課すと
非浪費性と PE は一般に達成できない.そこで,保護を制約に加えた非浪費性と PE を使う.
\begin{block}{Definition(制約付き非浪費性)}
  マッチング $Y$ において,学生 $i$ がカテゴリ $c$ を \textbf{strongly claim する}とは,
  \[
    c \in V(i), \qquad c \succ_i Y_i, \qquad
    (Y \setminus Y_i) \cup \{(i,c)\} \text{ が}
  \]
  quota feasibility・score-feasibility・reserved category protection
  を満たすこと.
  \\ strongly claim する学生が存在しないとき,$Y$ は制約付き非浪費性を満たす.
\end{block}
\begin{block}{Definition(制約付き PE)}
  quota feasibility・score-feasibility・reserved category protection を満たすマッチング $Y$ が
  \textbf{制約付き PE} $\iff$
  これら三つを満たす他のマッチングに Pareto 支配されない.
\end{block}
{\footnotesize
上の定義は保護に $\dtop{}$ を使うので「制約付き非浪費性($\dtop{}$)」「制約付き PE($\dtop{}$)」と書く.}
\end{frame}

\begin{frame}{二つの不可能性}
\begin{block}{Theorem 2(Reserved category protection と非浪費性)}
  quota feasibility・score-feasibility・非浪費性・reserved category protection をすべて満たす
  マッチングが存在しない場合がある.
\end{block}
\small
保護された $t$ の定員の分だけ $o$ の定員が減り,高得点者が落ちて $e_o$ が上がる
$\to$ $t$ を望む学生が $e_o-\delta$ に届かず,保護した $t$ の定員に誰も入らない.
\normalsize
\medskip
\begin{block}{Theorem 3(Reserved category protection と耐戦略性)}
  quota feasibility・score-feasibility・公平性・制約付き非浪費性・
  reserved category protection・耐戦略性を同時に満たすメカニズムは存在しない.
\end{block}
\small
$o\succ t$ の学生($80$ 点)が「$t$ が第一希望」と偽ると $\dtop{t}=1$ で $t$ の定員が確保され,
$t$ に割り当てられて得をする.
\end{frame}

\begin{frame}{Flexible SSD}
\small
どちらのメカニズムも SSD(点数の高い順に,選べる中で最も好むものに割り当てる)を使う.
違いは\textbf{各学生の番に選べるカテゴリの集合 $C_i$} だけである.
\begin{block}{Flexible SSD}
  現時点の部分マッチングを $Y$,cutoff を $e_o$ とする.
  \[
    A_i := \{\, c \in T \cap V(i) : |Y_c| < q_c,\ \sigma(i) \geq e_o - \delta \,\}
  \]
  \[
    C_i :=
    \begin{cases}
      \{\emptyset\} & \sum_{c \in V} |Y_c| = Q \\[4pt]
      \{\emptyset, o\} \cup A_i & \text{otherwise}
    \end{cases}
  \]
  $i$ を $C_i$ の中で $\succ_i$ について最も好むものに割り当てる.
\end{block}
$\Rightarrow$ 留保カテゴリの需要を考慮せず,未使用の定員はすべて $o$ に回してよい
\end{frame}

\begin{frame}{Reserved category protecting SSD}
\small
\begin{block}{Reserved category protecting SSD}
  Flexible SSD の $C_i$ から,次を満たさないときは $o$ を除く:
  \[
    |Y_o| \;<\; Q - \sum_{c \in T} \max\{\, |Y_c|,\ \dtop{c} \,\}
    \;=\; q_o + \sum_{c \in T} r_c(Y)
  \]
  $\Rightarrow$ $o$ に割り当てた後も reserved category protection が保たれるときだけ $o$ を選べる
\end{block}
\begin{block}{Theorem 4}
  Reserved category protecting SSD は,quota feasibility・score-feasibility・公平性・
  制約付き非浪費性・reserved category protection を満たす.\\
  さらに,\alert{制約付き PE} である.
\end{block}
Theorem 2 より PE は満たせない.
Theorem 3 より耐戦略性も満たさない.
\end{frame}

\begin{frame}{PE と制約付き PE}
\small
\begin{block}{PE}
  quota feasible かつ score-feasible なマッチングの中に,
  誰も悪くせずに誰かを良くするマッチングがない.
\end{block}
\begin{block}{制約付き PE}
  Pareto 支配するかを調べる比較対象を,\alert{reserved category protection も満たすマッチングだけ}に限定する.
\end{block}
\end{frame}

\begin{frame}{虚偽申告で得をする二つの例}
\small
\begin{block}{例 1:第一希望を偽る($\dtop{}$ の場合)}
  本当は $o\succ t$ の学生が「$t$ が第一希望」と申告する.
  $\dtop{t}$ が増えて $r_t$ が減り,$o$ の定員が減る.
  高得点の学生が $o$ から落ちて $e_o$ が上がるが,自分は確保された $t$ に入れる.
\end{block}
\begin{block}{例 2:応募可能なカテゴリを隠す}
  保護水準を「$c$ を $\emptyset$ より好む学生の人数」で定めたとする:
  \[
    \dacc{c} := \min\bigl\{\, q_c,\ |\{\, i \in I : c \succ_i \emptyset \,\}| \,\bigr\}
  \]
  本当は $o\succ t\succ\emptyset$ の学生が $t$ を受け入れないと申告すると,
  $\dacc{t}$ が減って $o$ の定員が増え,自分がより好む $o$ に入れる.
\end{block}
\medskip
どちらの例も,\textbf{学生の申告で保護水準 $d$ が変わる}ことが原因である.
そこで,保護水準を申告に依存しない情報で決めればよいのではないか.
\end{frame}

\begin{frame}{提案方式:eligibility-based reserved category protection}
\small
第一希望の人数ではなく\textbf{資格保有者の人数}で保護水準を定める.
\[
  \dV{c} := \min\bigl\{\, q_c,\ |\{\, i \in I : c \in V(i) \,\}| \,\bigr\}
\]
\begin{block}{Definition(Eligibility-based reserved category protection)}
  \[
    |Y_o| \;\leq\; q_o + \sum_{c \in T} \bigl( q_c - \max\{\, |Y_c|,\ \dV{c} \,\} \bigr)
  \]
\end{block}
\begin{block}{Proposition(現行の reserved category protection より強い)}
  $\topf_i = c \Rightarrow c \in V(i)$ より $\dV{c} \geq \dtop{c}$.
  右辺は保護水準について非増加なので,
  eligibility-based reserved category protection を満たすマッチングは,
  reserved category protection も満たす.
\end{block}
資格 $V(i)$ は大学が確認でき,学生は資格を隠せない.\\
つまり保護水準は選好にも申告にも依存せず,学生には変えられない.
\end{frame}

\begin{frame}{制約付き非浪費性($d^{V}$)と制約付き PE($d^{V}$)}
\small
reserved category protection を eligibility-based reserved category protection に置き換えて,
同じ概念を定める.
\begin{block}{Definition(制約付き非浪費性($d^{V}$))}
  マッチング $Y$ において,学生 $i$ がカテゴリ $c$ を \textbf{strongly claim する}とは,
  \[
    c \in V(i), \qquad c \succ_i Y_i, \qquad
    (Y \setminus Y_i) \cup \{(i,c)\} \text{ が}
  \]
  quota feasibility・score-feasibility・eligibility-based reserved category protection
  を満たすこと.
  \\ strongly claim する学生が存在しないとき,$Y$ は制約付き非浪費性($d^{V}$)を満たす.
\end{block}
\begin{block}{Definition(制約付き PE($d^{V}$))}
  quota feasibility・score-feasibility・eligibility-based reserved category protection を満たす
  マッチング $Y$ が \textbf{制約付き PE($d^{V}$)} $\iff$
  これら三つを満たす他のマッチングに Pareto 支配されない.
\end{block}
\end{frame}

\begin{frame}{固定カテゴリモデルへの帰着}
\begin{block}{固定カテゴリモデルへの帰着}
  柔軟カテゴリで quota feasibility と eligibility-based reserved category protection を満たす
  $\iff$
  $o$ の定員を
  \[
    q'_o := Q - \sum_{c \in T} \dV{c}
  \]
  各 $c \in T$ の定員を $q_c$ とした固定カテゴリモデルで quota feasibility を満たす
\end{block}
\begin{itemize}
  \item $q'_o$ は資格の情報だけで決まる\textbf{定数}(マッチングにも選好にも依存しない)
  \item \alert{Theorem 1(固定カテゴリでは SSD が全部満たす)をそのまま使える}
\end{itemize}
\end{frame}

\begin{frame}{Eligibility-based SSD のアルゴリズム}
\footnotesize
\textbf{入力}:学生 $I$,点数 $\sigma$,資格 $V(i)$,選好 $\succ_i$,定員 $(q_c)_{c \in V}$,$\delta$
\begin{enumerate}
  \item \textbf{保護水準の計算}\quad
        $\dV{c} = \min\{ q_c,\ |\{ i : c \in V(i) \}| \}\ \ (c \in T)$
  \item \textbf{$o$ の定員の計算}\quad $q'_o = Q - \sum_{c \in T} \dV{c}$
        \ (帰着より $o$ の定員として固定できる)
  \item \textbf{初期化}\quad $Y = \emptyset$,$e_o = 0$
  \item \textbf{点数の高い順に}学生 $i$ を一人ずつ処理する
  \begin{enumerate}
    \item 選べるカテゴリの集合を定める:
    \[
      A_i := \{\, c \in T \cap V(i) : |Y_c| < q_c,\ \sigma(i) \geq e_o - \delta \,\}
    \]
    \[
      C_i := \{\emptyset\} \ \cup\ \{\, o : |Y_o| < q'_o \,\} \ \cup\ A_i
    \]
    \item $C_i$ の中で $\succ_i$ について最も好む $c^*$ を選び,$Y := Y \cup \{(i, c^*)\}$
    \item $c^* = \emptyset$,$o \succ_i \emptyset$ で,
          $i$ が最初の脱落者なら $e_o := \sigma(i)$
  \end{enumerate}
  \item $Y$ を出力する
\end{enumerate}
\end{frame}

\begin{frame}{Eligibility-based SSD の動作例}
\small
$T = \{t\}$,$q_o = q_t = 1$($Q = 2$),$\delta = 10$.
\begin{center}
\renewcommand{\arraystretch}{1.1}
\begin{tabular}{cccc}
  \toprule
  学生 & $\sigma$ & $V(i)$ & 選好 \\
  \midrule
  $i_1$ & 100 & $\{o\}$ & $o \succ \emptyset$ \\
  $i_2$ & 90 & $\{o\}$ & $o \succ \emptyset$ \\
  $i_3$ & 80 & $\{o, t\}$ & $o \succ t \succ \emptyset$ \\
  \bottomrule
\end{tabular}
\end{center}
\textbf{Step 1--2}\quad $t$ の資格保有者は $i_3$ の 1 人なので $\dV{t} = 1$,$q'_o = 2 - 1 = 1$

\textbf{Step 4}
\begin{itemize}
  \item $i_1$:$C_{i_1} = \{\emptyset, o\}$ $\Rightarrow$ $i_1 \to o$
  \item $i_2$:$|Y_o| = q'_o$,$t \notin V(i_2)$ より $C_{i_2} = \{\emptyset\}$
        $\Rightarrow$ $i_2 \to \emptyset$,$e_o = 90$
  \item $i_3$:$80 \geq 90 - 10$ より $C_{i_3} = \{\emptyset, t\}$
        $\Rightarrow$ $i_3 \to t$
\end{itemize}

(現行の reserved category protection では,正直に申告すると $\dtop{t} = 0$ で $i_3$ は不合格になる:Theorem 3 の例)
\end{frame}

\begin{frame}{Eligibility-based SSD の性質}
\small
\begin{block}{メカニズム}
  $o$ の定員を $q'_o$,各 $c \in T$ の定員を $q_c$ とした固定カテゴリモデルで SSD を実行する.
\end{block}
\begin{block}{Theorem 5}
  Eligibility-based SSD は,quota feasibility・score-feasibility・公平性・
  eligibility-based reserved category protection・制約付き非浪費性($d^{V}$)・
  \textbf{耐戦略性}を満たす.\\
  さらに,\alert{制約付き PE($d^{V}$)}である.
\end{block}
\textbf{証明の概要}
\begin{enumerate}
  \item 帰着より,問題は定員 $(q'_o, (q_c)_{c\in T})$ の固定カテゴリモデルと同じ
  \item Theorem 1 より,SSD は quota feasibility・score-feasibility・公平性・PE・耐戦略性を満たす
  \item 帰着後のモデルで quota feasible なマッチングの集合は,
        元のモデルで quota feasibility と eligibility-based reserved category protection を満たすマッチングの集合と一致する
\end{enumerate}
\end{frame}
